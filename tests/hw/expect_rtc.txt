# The real-time clock (rtc_9.c, rtc_7.c). The clock starts at --rtc and runs on emulated time;
# the ARM7 reads it at frame 61, a little over a second in, so one second has passed.
args: --rtc 2026-11-07T12:34:56
# date, day of week (Sunday 0), time; status 1 (24-hour mode, no flags) and 2
rtc 2026-11-07 6 12:34:57
rtcstat 02 00
# 2099-12-31 23:59:59 written and read back, and its hour in 12-hour mode: 11 with the PM bit
rtcwrite 2099-12-31 4 23:59:59 51
rtcirq ok
